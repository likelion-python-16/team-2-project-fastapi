```python
# prepare_data.py
from datasets import load_dataset, Dataset, concatenate_datasets

def load_and_map(name, text_col, label_col, label_map=None, split="train"):
    ds = load_dataset(name, split=split)
    if text_col != "text": ds = ds.rename_columns({text_col:"text"})
    if label_col != "label": ds = ds.rename_columns({label_col:"label"})
    if label_map:  # 문자열 라벨 → 정수(0=clean, 1=toxic)
        def _m(x):
            v = x["label"]
            if isinstance(v, str): v = label_map.get(v.lower().strip(), 0)
            x["label"] = int(v)
            return x
        ds = ds.map(_m)
    return ds.remove_columns([c for c in ds.column_names if c not in ("text","label")])

# 1) MIT: 한국어 악성댓글
ds_km = load_and_map("korean-malicious-comments-dataset",
                     text_col="comment", label_col="label",
                     label_map={"clean":0,"toxic":1})

# 2) MIT: Curse-detection-data
ds_curse = load_and_map("Curse-detection-data",
                        text_col="text", label_col="label",
                        label_map={"normal":0,"clean":0,"curse":1,"toxic":1,"offensive":1})

# 3) Apache-2.0: HateScore (컬럼명은 저장본에 따라 다를 수 있음)
try:
    ds_hs = load_and_map("hatescore-korean-hate-speech",
                         text_col="comment", label_col="label",
                         label_map={"clean":0,"non-toxic":0,"toxic":1,"offensive":1,"hate":1})
except Exception:
    ds_hs = None  # 로딩 실패 시 건너뜀

# 4) CC0: NSMC (감성) → positive/negative가 아니라서 "클린 다양화" 용도로 일부만 0으로 사용
ds_nsmc = load_dataset("nsmc", split="train")  # columns: id, document, label(0/1: 1=긍정)
ds_nsmc = ds_nsmc.rename_columns({"document":"text"})[["text","label"]]
# 긍정/부정 상관없이, 과다 반영 금지. 단, 여기선 간단히 'clean=0'으로 통일해 소량만 사용.
ds_nsmc_clean = ds_nsmc.shuffle(seed=42).select(range(20000)).map(lambda x: {"label":0})

parts = [ds_km, ds_curse, ds_nsmc_clean] + ([ds_hs] if ds_hs else [])
train_all = concatenate_datasets(parts).shuffle(seed=42)
splits = train_all.train_test_split(test_size=0.1, seed=42)
splits.save_to_disk("./_data_toxic_bin")
print(splits)
```

> 팁: **NSMC 비율을 과하게 넣지 말기**(유해/비유해 클래스 밸런스 무너질 수 있음). 처음엔 1~2만 샘플 정도로 충분.

# 3) 파인튜닝(klue/roberta-base 권장)

```python
# train.py
from datasets import load_from_disk
from transformers import AutoTokenizer, AutoModelForSequenceClassification, TrainingArguments, Trainer
import numpy as np, evaluate

splits = load_from_disk("./_data_toxic_bin")
model_name = "klue/roberta-base"  # 또는 monologg/kobert (허깅페이스로 욕설 감지 필요  kobert-수위판별모델 (NSFW + 욕설 탐지 Ai)
tok = AutoTokenizer.from_pretrained(model_name)

def tok_fn(b): return tok(b["text"], truncation=True, padding="max_length", max_length=160)
enc = splits.map(tok_fn, batched=True)

id2label={0:"clean",1:"toxic"}; label2id={"clean":0,"toxic":1}
model = AutoModelForSequenceClassification.from_pretrained(model_name, num_labels=2,
                                                           id2label=id2label, label2id=label2id)

acc = evaluate.load("accuracy"); f1 = evaluate.load("f1"); prec=evaluate.load("precision"); rec=evaluate.load("recall")
def metrics(p):
    pred = np.argmax(p.predictions, axis=1)
    return {
        "acc": acc.compute(predictions=pred, references=p.label_ids)["accuracy"],
        "f1":  f1.compute(predictions=pred, references=p.label_ids, average="binary")["f1"],
        "precision": prec.compute(predictions=pred, references=p.label_ids, average="binary")["precision"],
        "recall":    rec.compute(predictions=pred, references=p.label_ids, average="binary")["recall"],
    }

args = TrainingArguments(
    output_dir="./toxic-roberta",
    learning_rate=2e-5,
    num_train_epochs=2,             # 시간/자원에 맞게 2~4
    per_device_train_batch_size=16,
    per_device_eval_batch_size=32,
    evaluation_strategy="epoch",
    save_strategy="epoch",
    metric_for_best_model="f1",
    load_best_model_at_end=True,
    fp16=True
)

trainer = Trainer(model=model, args=args,
                  train_dataset=enc["train"], eval_dataset=enc["test"],
                  tokenizer=tok, compute_metrics=metrics)
trainer.train()
trainer.save_model("./toxic-roberta"); tok.save_pretrained("./toxic-roberta")
```

# 4) 임계치 정책 + FastAPI 서빙

```python
# serve.py
from fastapi import FastAPI
from transformers import AutoTokenizer, AutoModelForSequenceClassification, TextClassificationPipeline

MODEL_DIR = "./toxic-roberta"
tok  = AutoTokenizer.from_pretrained(MODEL_DIR)
mdl  = AutoModelForSequenceClassification.from_pretrained(MODEL_DIR)
pipe = TextClassificationPipeline(model=mdl, tokenizer=tok, top_k=None)

THRESH_BLOCK, THRESH_REVIEW = 0.85, 0.70
app = FastAPI(title="Korean Toxicity API")

@app.post("/moderate")
def moderate(text: str):
    out = pipe(text, truncation=True, max_length=160)
    # 예시: [{'label':'toxic','score':0.93}, {'label':'clean','score':0.07}]
    toxic = max((d["score"] for d in out if d["label"].lower()=="toxic"), default=0.0)
    decision = "block" if toxic>=THRESH_BLOCK else ("review" if toxic>=THRESH_REVIEW else "allow")
    return {"score":round(toxic,3), "decision":decision, "labels":out}
```

```python
# guard.py
import re
BAD = {"씨발","ㅅㅂ","좆","개새끼"}  # 테스트용 소량 (출처 불명확 리스트는 '룰'로만 사용, '학습'엔 넣지 말기)
PHONE_RE = re.compile(r'\b(01[0-9]-?\d{3,4}-?\d{4})\b')
EMAIL_RE = re.compile(r'\b[A-Za-z0-9._%+-]+@[A-Za-z0-9.-]+\.[A-Za-z]{2,}\b')
URL_RE   = re.compile(r'https?://\S+')

def normalize(s):  # 특수문자 제거/반복 축약
    s = re.sub(r'[^가-힣a-zA-Z0-9 ]','', s.lower())
    s = re.sub(r'(.)\1{2,}', r'\1\1', s)
    return s

def rule_hit(s): return any(b in normalize(s) for b in BAD)
def mask_pii(s):
    s = PHONE_RE.sub("[전화번호]", s)
    s = EMAIL_RE.sub("[이메일]", s)
    s = URL_RE.sub("[링크]", s)
    return s
```

```python
from sqlalchemy.orm import Session
```

# 5) app/export_reports.py — **CSV로 내보내기**

- `report_type='abuse'` 만 추출 (필요하면 reason LIKE '%욕설%' 조건으로 바꾸면 됨)
    
- 거짓 신고는 `label=0`로, 일반 욕설 신고는 `label=1`로 저장(선택 규칙)
    

```python
import os
import pandas as pd
from sqlalchemy.orm import Session
from .db import SessionLocal
from .models import Report

EXPORT_CSV = os.getenv("EXPORT_CSV", "./data/reported_samples.csv")

def export_reports_to_csv() -> int:
    os.makedirs(os.path.dirname(EXPORT_CSV), exist_ok=True)
    db: Session = SessionLocal()
    try:
        q = (
            db.query(Report)
            .filter(Report.report_type == "abuse")      # 욕설 신고만
            .order_by(Report.created_at.desc())
        )
        rows = q.all()
        data = []
        for r in rows:
            text = (r.content or "").strip()
            if not text:
                continue
            # 거짓 신고는 label=0, 일반 신고는 1 (정책에 맞게 조정해도 됨)
            label = 0 if r.is_false_report else 1
            data.append({"text": text, "label": label})
        df = pd.DataFrame(data)
        df.to_csv(EXPORT_CSV, index=False, encoding="utf-8-sig")
        print(f"📄 Exported {len(df)} rows → {EXPORT_CSV}")
        return len(df)
    finally:
        db.close()

if __name__ == "__main__":
    export_reports_to_csv()
```

# 6) app/retrain.py — **재학습 + 모델 교체**

- 학습 데이터: **안전 라이선스 묶음** + `reported_samples.csv`
    
- 기존 모델(`MODEL_DIR`)에서 이어서 1 epoch 빠르게 미세조정 → `NEW_MODEL_DIR`로 저장
    

```python
import os
import pandas as pd
from datasets import load_dataset, load_from_disk, Dataset, concatenate_datasets
from transformers import AutoTokenizer, AutoModelForSequenceClassification, TrainingArguments, Trainer

EXPORT_CSV = os.getenv("EXPORT_CSV", "./data/reported_samples.csv")
MODEL_DIR  = os.getenv("MODEL_DIR", "./toxic-roberta")
NEW_MODEL_DIR = os.getenv("NEW_MODEL_DIR", "./toxic-roberta-v2")

SAFE_SPLITS_PATH = "./_data_toxic_bin"  # 이전에 준비해둔 안전 데이터(2~단계에서 만든 것)

def load_safe_splits():
    # 없다면 prepare_data.py 같은 걸로 먼저 생성해두세요.
    return load_from_disk(SAFE_SPLITS_PATH)

def load_report_csv():
    if not os.path.exists(EXPORT_CSV):
        return None
    df = pd.read_csv(EXPORT_CSV)
    df = df.dropna(subset=["text","label"])
    df["label"] = df["label"].astype(int).clip(0,1)
    return Dataset.from_pandas(df[["text","label"]])

def retrain_once():
    splits = load_safe_splits()
    ds_reports = load_report_csv()

    # 학습 데이터 합치기(신고 데이터가 있으면 train에 합침)
    train_ds = splits["train"]
    if ds_reports and len(ds_reports) > 0:
        train_ds = concatenate_datasets([train_ds, ds_reports]).shuffle(seed=43)

    tok = AutoTokenizer.from_pretrained(MODEL_DIR)
    def enc(b): return tok(b["text"], truncation=True, padding="max_length", max_length=160)
    enc_train = train_ds.map(enc, batched=True)
    enc_eval  = splits["test"].map(enc, batched=True)

    model = AutoModelForSequenceClassification.from_pretrained(MODEL_DIR)

    args = TrainingArguments(
        output_dir=NEW_MODEL_DIR,
        num_train_epochs=1,                     # 빠르게 1 에폭
        per_device_train_batch_size=16,
        per_device_eval_batch_size=32,
        evaluation_strategy="epoch",
        save_strategy="epoch",
        learning_rate=2e-5,
        fp16=True,
        report_to="none"
    )
    trainer = Trainer(model=model, args=args,
                      train_dataset=enc_train, eval_dataset=enc_eval, tokenizer=tok)
    trainer.train()
    trainer.save_model(NEW_MODEL_DIR); tok.save_pretrained(NEW_MODEL_DIR)
    print(f"✅ Saved new model → {NEW_MODEL_DIR}")

if __name__ == "__main__":
    retrain_once()
```

# 7) app/serve.py — **서빙 + 핫스왑(엔드포인트로 모델 교체)**

```python
import os
from fastapi import FastAPI
from transformers import AutoTokenizer, AutoModelForSequenceClassification, TextClassificationPipeline

MODEL_DIR = os.getenv("MODEL_DIR", "./toxic-roberta")
NEW_MODEL_DIR = os.getenv("NEW_MODEL_DIR", "./toxic-roberta-v2")

def load_pipe(path:str):
    tok = AutoTokenizer.from_pretrained(path)
    mdl = AutoModelForSequenceClassification.from_pretrained(path)
    return TextClassificationPipeline(model=mdl, tokenizer=tok, top_k=None)

pipe = load_pipe(MODEL_DIR)

THRESH_BLOCK, THRESH_REVIEW = 0.85, 0.70
app = FastAPI(title="Moderation API")

@app.post("/moderate")
def moderate(text: str):
    out = pipe(text, truncation=True, max_length=160)
    toxic = max((d["score"] for d in out if d["label"].lower()=="toxic"), default=0.0)
    decision = "block" if toxic>=THRESH_BLOCK else ("review" if toxic>=THRESH_REVIEW else "allow")
    return {"score": round(toxic,3), "decision": decision, "labels": out}

@app.post("/reload-model")
def reload_model():
    global pipe, MODEL_DIR
    # NEW_MODEL_DIR의 모델을 현재 서빙 모델로 교체
    pipe = load_pipe(NEW_MODEL_DIR)
    MODEL_DIR = NEW_MODEL_DIR
    return {"ok": True, "loaded": MODEL_DIR}
```

실행:

`uvicorn app.serve:app --reload`

# 8) (선택) app/guard.py — 룰 + PII 마스킹

```PYTHON
import re
BAD={"씨발","ㅅㅂ","좆","개새끼"}  # 운영 시엔 출처 불명확 리스트는 '학습X, 룰전용O'
PHONE_RE=re.compile(r'\b(01[0-9]-?\d{3,4}-?\d{4})\b')
EMAIL_RE=re.compile(r'\b[A-Za-z0-9._%+-]+@[A-Za-z0-9.-]+\.[A-Za-z]{2,}\b')
URL_RE=re.compile(r'https?://\S+')

def normalize(s):
    s=re.sub(r'[^가-힣a-zA-Z0-9 ]','',s.lower())
    s=re.sub(r'(.)\1{2,}', r'\1\1', s)  # 반복 축약
    return s

def rule_hit(s): return any(b in normalize(s) for b in BAD)
def mask_pii(s):
    s=PHONE_RE.sub("[전화번호]",s); s=EMAIL_RE.sub("[이메일]",s); s=URL_RE.sub("[링크]",s)
    return s
```
